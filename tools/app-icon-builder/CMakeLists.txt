cmake_minimum_required(VERSION 3.21)
project(AppIconBuilder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Gui)
add_executable(app_icon_builder main.cpp)
target_link_libraries(app_icon_builder PRIVATE Qt6::Gui)
if(MSVC)
    target_compile_options(app_icon_builder PRIVATE /W4 /utf-8)
endif()
if(WIN32)
    target_compile_definitions(app_icon_builder PRIVATE UNICODE _UNICODE)
    target_compile_definitions(app_icon_builder PRIVATE UNICODE _UNICODE)
    target_link_libraries(app_icon_builder PRIVATE user32 gdi32)
endif()
