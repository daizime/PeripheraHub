cmake_minimum_required(VERSION 3.21)
project(FluentIconRenderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Svg)
add_executable(fluent_icon_renderer main.cpp)
target_link_libraries(fluent_icon_renderer PRIVATE Qt6::Core Qt6::Gui Qt6::Svg)
if(MSVC)
    target_compile_options(fluent_icon_renderer PRIVATE /W4 /utf-8)
endif()
